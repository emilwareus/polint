24/// Flags indicating whether a memory access modifies or references memory.
25///
26/// This is no access at all, a modification, a reference, or both
27/// a modification and a reference.
28enum class ModRefInfo : uint8_t {
29 /// The access neither references nor modifies the value stored in memory.
30 NoModRef = 0,
31 /// The access may reference the value stored in memory.
32 Ref = 1,
33 /// The access may modify the value stored in memory.
34 Mod = 2,
35 /// The access may reference and may modify the value stored in memory.
36 ModRef = Ref | Mod,
37 LLVM_MARK_AS_BITMASK_ENUM(ModRef),
38};
39
40[[nodiscard]] inline bool isNoModRef(const ModRefInfo MRI) {
41 return MRI == ModRefInfo::NoModRef;
42}
43[[nodiscard]] inline bool isModOrRefSet(const ModRefInfo MRI) {
44 return MRI != ModRefInfo::NoModRef;
45}
46[[nodiscard]] inline bool isModAndRefSet(const ModRefInfo MRI) {
47 return MRI == ModRefInfo::ModRef;
48}
49[[nodiscard]] inline bool isModSet(const ModRefInfo MRI) {
50 return static_cast<int>(MRI) & static_cast<int>(ModRefInfo::Mod);
51}
52[[nodiscard]] inline bool isRefSet(const ModRefInfo MRI) {
53 return static_cast<int>(MRI) & static_cast<int>(ModRefInfo::Ref);
54}

59/// The locations at which a function might access memory.
60enum class IRMemLocation {
61 /// Access to memory via argument pointers.
62 ArgMem = 0,
63 /// Memory that is inaccessible via LLVM IR.
64 InaccessibleMem = 1,
65 /// Errno memory.
66 ErrnoMem = 2,
67 /// Any other memory.
68 Other = 3,
69 /// Represents target specific state.
70 TargetMem0 = 4,
71 TargetMem1 = 5,
72
73 /// Helpers to iterate all locations in the MemoryEffectsBase class.
74 First = ArgMem,
75 Last = TargetMem1,
76};

78template <typename LocationEnum> class MemoryEffectsBase {
79public:
80 using Location = LocationEnum;
81
82private:
83 uint32_t Data = 0;
84
85 static constexpr uint32_t BitsPerLoc = 2;
86 static constexpr uint32_t LocMask = (1 << BitsPerLoc) - 1;
87
88 static uint32_t getLocationPos(Location Loc) {
89 return (uint32_t)Loc * BitsPerLoc;
90 }
91
92 MemoryEffectsBase(uint32_t Data) : Data(Data) {}
93
94 void setModRef(Location Loc, ModRefInfo MR) {
95 Data &= ~(LocMask << getLocationPos(Loc));
96 Data |= static_cast<uint32_t>(MR) << getLocationPos(Loc);
97 }
98
99public:
100 /// Returns iterator over all supported location kinds.
101 static auto locations() {
102 return enum_seq_inclusive(Location::First, Location::Last,
103 force_iteration_on_noniterable_enum);
104 }
105 /// Returns iterator over all target location kinds
106 static auto targetMemLocations() {
107 return enum_seq_inclusive(Location::TargetMem0, Location::TargetMem1,
108 force_iteration_on_noniterable_enum);
109 }
110
111 /// Create MemoryEffectsBase that can access only the given location with the
112 /// given ModRefInfo.
113 MemoryEffectsBase(Location Loc, ModRefInfo MR) { setModRef(Loc, MR); }
114
115 /// Create MemoryEffectsBase that can access any location with the given
116 /// ModRefInfo.
117 explicit MemoryEffectsBase(ModRefInfo MR) {
118 for (Location Loc : locations())
119 setModRef(Loc, MR);
120 }
121
122 /// Create MemoryEffectsBase that can read and write any memory.
123 static MemoryEffectsBase unknown() {
124 return MemoryEffectsBase(ModRefInfo::ModRef);
125 }
126
127 /// Create MemoryEffectsBase that cannot read or write any memory.
128 static MemoryEffectsBase none() {
129 return MemoryEffectsBase(ModRefInfo::NoModRef);
130 }
131
132 /// Create MemoryEffectsBase that can read any memory.
133 static MemoryEffectsBase readOnly() {
134 return MemoryEffectsBase(ModRefInfo::Ref);
135 }
136
137 /// Create MemoryEffectsBase that can write any memory.
138 static MemoryEffectsBase writeOnly() {
139 return MemoryEffectsBase(ModRefInfo::Mod);
140 }
141
142 /// Create MemoryEffectsBase that can only access argument memory.
143 static MemoryEffectsBase argMemOnly(ModRefInfo MR = ModRefInfo::ModRef) {
144 return MemoryEffectsBase(Location::ArgMem, MR);
145 }
146
147 /// Create MemoryEffectsBase that can only access inaccessible memory.

151 }
152
153 /// Create MemoryEffectsBase that can only access errno memory.
154 static MemoryEffectsBase errnoMemOnly(ModRefInfo MR = ModRefInfo::ModRef) {
155 return MemoryEffectsBase(Location::ErrnoMem, MR);
156 }
157
158 /// Create MemoryEffectsBase that can only access other memory.

160 return MemoryEffectsBase(Location::Other, MR);

163 /// Create MemoryEffectsBase that can only access inaccessible or argument
164 /// memory.
165 static MemoryEffectsBase
166 inaccessibleOrArgMemOnly(ModRefInfo MR = ModRefInfo::ModRef) {
167 MemoryEffectsBase FRMB = none();
168 FRMB.setModRef(Location::ArgMem, MR);

218 /// Get ModRefInfo for the given Location.
219 ModRefInfo getModRef(Location Loc) const {
220 return ModRefInfo((Data >> getLocationPos(Loc)) & LocMask);
221 }

223 /// Get new MemoryEffectsBase with modified ModRefInfo for Loc.

226 ME.setModRef(Loc, MR);
227 return ME;

230 /// Get new MemoryEffectsBase with NoModRef on the given Loc.

237 /// Get ModRefInfo for any location.
238 ModRefInfo getModRef() const {
239 ModRefInfo MR = ModRefInfo::NoModRef;

242 return MR;

245 /// Whether this function accesses no memory.
246 bool doesNotAccessMemory() const { return Data == 0; }
247
248 /// Whether this function only (at most) reads memory.
249 bool onlyReadsMemory() const { return !isModSet(getModRef()); }
250
251 /// Whether this function only (at most) writes memory.
252 bool onlyWritesMemory() const { return !isRefSet(getModRef()); }
253
254 /// Whether this function only (at most) accesses argument memory.
255 bool onlyAccessesArgPointees() const {
256 return getWithoutLoc(Location::ArgMem).doesNotAccessMemory();
257 }
258
259 /// Whether this function may access argument memory.
260 bool doesAccessArgPointees() const {
261 return isModOrRefSet(getModRef(Location::ArgMem));
262 }
263
264 /// Whether this function only (at most) accesses inaccessible memory.
265 bool onlyAccessesInaccessibleMem() const {
266 return getWithoutLoc(Location::InaccessibleMem).doesNotAccessMemory();
267 }
268
269 /// Whether this function only (at most) accesses inaccessible or target
270 /// memory.
271 bool onlyAccessesInaccessibleOrTargetMem() const {
272 MemoryEffectsBase ME = *this;
273 for (auto Loc : MemoryEffectsBase::targetMemLocations())
274 ME &= ME.getWithoutLoc(Loc);
275 return ME.getWithoutLoc(Location::InaccessibleMem).doesNotAccessMemory();
276 }
277
278 /// Whether location is target memory location.
279 bool isTargetMemLoc(IRMemLocation Loc) const {
280 for (auto L : targetMemLocations())
281 if (Loc == L)
282 return true;
283 return false;
284 }
285
286 /// Whether the target memory locations are all the same.
287 /// So it behaves as the default read/write, but for Target
288 /// locations only.
289 bool isTargetMemLocSameForAll() const {
290 ModRefInfo Expected = getModRef(IRMemLocation::TargetMem0);
291 for (auto Loc : targetMemLocations()) {
292 if (Expected != getModRef(Loc))
293 return false;
294 }
295 return true;
296 }
297
298 /// Whether this function only (at most) accesses errno memory.
299 bool onlyAccessesErrnoMem() const {
300 return getWithoutLoc(Location::ErrnoMem).doesNotAccessMemory();
301 }
302
303 /// Whether this function only (at most) accesses argument and inaccessible
304 /// memory.
305 bool onlyAccessesInaccessibleOrArgMem() const {
306 return getWithoutLoc(Location::InaccessibleMem)
307 .getWithoutLoc(Location::ArgMem)
308 .doesNotAccessMemory();
309 }

311 /// Intersect with other MemoryEffectsBase.
312 MemoryEffectsBase operator&(MemoryEffectsBase Other) const {
313 return MemoryEffectsBase(Data & Other.Data);
314 }
315
316 /// Intersect (in-place) with other MemoryEffectsBase.
317 MemoryEffectsBase &operator&=(MemoryEffectsBase Other) {
318 Data &= Other.Data;
319 return *this;
320 }
321
322 /// Union with other MemoryEffectsBase.
323 MemoryEffectsBase operator|(MemoryEffectsBase Other) const {
324 return MemoryEffectsBase(Data | Other.Data);
325 }
326
327 /// Union (in-place) with other MemoryEffectsBase.
328 MemoryEffectsBase &operator|=(MemoryEffectsBase Other) {
329 Data |= Other.Data;
330 return *this;
331 }
332
333 /// Subtract other MemoryEffectsBase.
334 MemoryEffectsBase operator-(MemoryEffectsBase Other) const {
335 return MemoryEffectsBase(Data & ~Other.Data);
336 }
337
338 /// Subtract (in-place) with other MemoryEffectsBase.
339 MemoryEffectsBase &operator-=(MemoryEffectsBase Other) {
340 Data &= ~Other.Data;
341 return *this;
342 }
343
344 /// Check whether this is the same as other MemoryEffectsBase.
345 bool operator==(MemoryEffectsBase Other) const { return Data == Other.Data; }
346
347 /// Check whether this is different from other MemoryEffectsBase.
348 bool operator!=(MemoryEffectsBase Other) const { return !operator==(Other); }
349};

364/// Components of the pointer that may be captured.
365enum class CaptureComponents : uint8_t {
366 None = 0,
367 AddressIsNull = (1 << 0),
368 Address = (1 << 1) | AddressIsNull,
369 ReadProvenance = (1 << 2),
370 Provenance = (1 << 3) | ReadProvenance,
371 All = Address | Provenance,
372 LLVM_MARK_AS_BITMASK_ENUM(Provenance),
373};
374
375inline bool capturesNothing(CaptureComponents CC) {
376 return CC == CaptureComponents::None;
377}
378
379inline bool capturesAnything(CaptureComponents CC) {
380 return CC != CaptureComponents::None;
381}
382
383inline bool capturesAddressIsNullOnly(CaptureComponents CC) {
384 return (CC & CaptureComponents::Address) == CaptureComponents::AddressIsNull;
385}
386
387inline bool capturesAddress(CaptureComponents CC) {
388 return (CC & CaptureComponents::Address) != CaptureComponents::None;
389}
390
391inline bool capturesReadProvenanceOnly(CaptureComponents CC) {
392 return (CC & CaptureComponents::Provenance) ==
393 CaptureComponents::ReadProvenance;
394}
395
396inline bool capturesFullProvenance(CaptureComponents CC) {
397 return (CC & CaptureComponents::Provenance) == CaptureComponents::Provenance;
398}
399
400inline bool capturesAnyProvenance(CaptureComponents CC) {
401 return (CC & CaptureComponents::Provenance) != CaptureComponents::None;
402}
403
404inline bool capturesAll(CaptureComponents CC) {
405 return CC == CaptureComponents::All;
406}

410/// Represents which components of the pointer may be captured in which
411/// location. This represents the captures(...) attribute in IR.
412///
413/// For more information on the precise semantics see LangRef.
414class CaptureInfo {
415 CaptureComponents OtherComponents;
416 CaptureComponents RetComponents;
417
418public:
419 CaptureInfo(CaptureComponents OtherComponents,
420 CaptureComponents RetComponents)
421 : OtherComponents(OtherComponents), RetComponents(RetComponents) {}
422
423 CaptureInfo(CaptureComponents Components)
424 : OtherComponents(Components), RetComponents(Components) {}
425
426 /// Create CaptureInfo that does not capture any components of the pointer
427 static CaptureInfo none() { return CaptureInfo(CaptureComponents::None); }
428
429 /// Create CaptureInfo that may capture all components of the pointer.
430 static CaptureInfo all() { return CaptureInfo(CaptureComponents::All); }
431
432 /// Create CaptureInfo that may only capture via the return value.
433 static CaptureInfo
434 retOnly(CaptureComponents RetComponents = CaptureComponents::All) {
435 return CaptureInfo(CaptureComponents::None, RetComponents);
436 }
437
438 /// Whether the pointer is only captured via the return value.
439 bool isRetOnly() const { return capturesNothing(OtherComponents); }
440
441 /// Get components potentially captured by the return value.
442 CaptureComponents getRetComponents() const { return RetComponents; }
443
444 /// Get components potentially captured through locations other than the
445 /// return value.
446 CaptureComponents getOtherComponents() const { return OtherComponents; }
447
448 /// Get the potentially captured components of the pointer (regardless of
449 /// location).
450 operator CaptureComponents() const { return OtherComponents | RetComponents; }
451
452 bool operator==(CaptureInfo Other) const {
453 return OtherComponents == Other.OtherComponents &&
454 RetComponents == Other.RetComponents;
455 }
456
457 bool operator!=(CaptureInfo Other) const { return !(*this == Other); }
458
459 /// Compute union of CaptureInfos.
460 CaptureInfo operator|(CaptureInfo Other) const {
461 return CaptureInfo(OtherComponents | Other.OtherComponents,
462 RetComponents | Other.RetComponents);
463 }
464
465 /// Compute intersection of CaptureInfos.
466 CaptureInfo operator&(CaptureInfo Other) const {
467 return CaptureInfo(OtherComponents & Other.OtherComponents,
468 RetComponents & Other.RetComponents);
469 }
470
471 /// Compute union of CaptureInfos in-place.
472 CaptureInfo &operator|=(CaptureInfo Other) {
473 OtherComponents |= Other.OtherComponents;
474 RetComponents |= Other.RetComponents;
475 return *this;
476 }
477
478 /// Compute intersection of CaptureInfos in-place.
479 CaptureInfo &operator&=(CaptureInfo Other) {
480 OtherComponents &= Other.OtherComponents;
481 RetComponents &= Other.RetComponents;
482 return *this;
483 }
484
485 static CaptureInfo createFromIntValue(uint32_t Data) {
486 return CaptureInfo(CaptureComponents(Data >> 4),
487 CaptureComponents(Data & 0xf));
488 }
489
490 /// Convert CaptureInfo into an encoded integer value (used by captures
491 /// attribute).
492 uint32_t toIntValue() const {
493 return (uint32_t(OtherComponents) << 4) | uint32_t(RetComponents);
494 }
495};
